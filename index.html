doc.moveDown(2); doc.fontSize(12).fillColor('#000000').text(content, { align: 'justify', lineGap: 5 }); const bottom = doc.page.height - 50; doc.fontSize(10).fillColor('#999999').text('Karya JOY Corporation', 50, bottom, { align: 'center' }); doc.end(); }); } app.get('/api/products', async (req, res) => { const { data } = await supabase.from('products').select('*').order('created_at', { ascending: false }); res.json(data || []); }); app.delete('/api/products/:id', async (req, res) => { const { data: product } = await supabase.from('products').select('file_path').eq('id', req.params.id).single(); if (product && product.file_path) await supabase.storage.from('joycorp-files').remove([product.file_path]); await supabase.from('products').delete().eq('id', req.params.id); res.json({ success: true }); }); app.get('/api/download-by-id/:id', async (req, res) => { const { data: product } = await supabase.from('products').select('file_path').eq('id', req.params.id).single(); if (!product || !product.file_path) return res.status(404).send('File tidak ada.'); const { data: urlData } = await supabase.storage.from('joycorp-files').createSignedUrl(product.file_path, 60); res.redirect(urlData.signedUrl); }); app.post('/api/factory/run', async (req, res) => { const { marketMode } = req.body; try { const prompt = `Buatkan JSON 4 produk digital BERVARIASI (1 E-Book, 1 Video Script, 1 Template, 1 Bot Script). Mode Market: ${marketMode || 'Trend Jacking'}. Format: {"products": [{"name":"...", "category":"...", "price":15, "description":"..."}]}`; const response = await fetch('https://api.groq.com/openai/v1/chat/completions', { method: 'POST', headers: { 'Authorization': `Bearer ${GROQ_API_KEY}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ model: 'llama3-8b-8192', messages: [{ role: 'user', content: prompt }], response_format: { type: "json_object" } }) }); const aiData = await response.json(); let parsed = JSON.parse(aiData.choices[0].message.content); const items = parsed.products || parsed; const validItems = []; for (const item of items) { if (!item.name || !item.description) continue; } await supabase.from('products').insert(validItems); res.json({ success: true, message: `${validItems.length} Produk dibuat.` }); } catch (err) { res.status(500).json({ error: "Pabrik gagal." }); } }); // Saldo Real (Mulai dari $1000) app.get('/api/sales', async (req, res) => { const { data: orders } = await supabase.from('orders').select('amount, created_at, id').eq('status', 'paid'); let total = 1000;
